<!-- ── SCRIPT ─────────────────────────────────────── -->
<script>
(function () {
  'use strict';

  /* ── Config ─────────────────────────────────── */
  const API_URL     = '/chat';          // Change to full URL if backend is on a different origin
  const OPEN_MSG    = 'Ciao! Sono Alessia, l\'assistente legale di LexFlow. Dimmi brevemente la tua questione e ti aiuto a inquadrarla.';
  const ERROR_MSG   = 'Si è verificato un errore. Riprova tra un momento.';

  /* ── Elements ────────────────────────────────── */
  const trigger  = document.getElementById('alessiaTrigger');
  const panel    = document.getElementById('alessiaChatPanel');
  const closeBtn = document.getElementById('alessiaClose');
  const messages = document.getElementById('alessiaMessages');
  const form     = document.getElementById('alessiaForm');
  const input    = document.getElementById('alessiaInput');
  const typing   = document.getElementById('alessiaTyping');
  const sendBtn  = form.querySelector('.alessia-send');

  /* ── State ───────────────────────────────────── */
  let isOpen       = false;
  let isWaiting    = false;
  let greeted      = false;
  let history      = [];   // [{role:'user'|'assistant', content:'...'}]

  /* ── Open / close ────────────────────────────── */
  function openPanel() {
    isOpen = true;
    panel.classList.add('is-open');
    panel.setAttribute('aria-hidden', 'false');
    trigger.setAttribute('aria-expanded', 'true');
    if (!greeted) { appendBot(OPEN_MSG); greeted = true; }
    input.focus();
  }

  function closePanel() {
    isOpen = false;
    panel.classList.remove('is-open');
    panel.setAttribute('aria-hidden', 'true');
    trigger.setAttribute('aria-expanded', 'false');
    trigger.focus();
  }

  trigger.addEventListener('click', () => isOpen ? closePanel() : openPanel());
  closeBtn.addEventListener('click', closePanel);

  /* Close on Escape */
  document.addEventListener('keydown', e => {
    if (e.key === 'Escape' && isOpen) closePanel();
  });

  /* ── Bubble helpers ──────────────────────────── */
  function appendBubble(text, role) {
    const div = document.createElement('div');
    div.className = 'ale-bubble ale-bubble--' + role;
    div.textContent = text;
    messages.appendChild(div);
    messages.scrollTop = messages.scrollHeight;
    return div;
  }
  const appendBot   = t => appendBubble(t, 'bot');
  const appendUser  = t => appendBubble(t, 'user');
  const appendError = t => appendBubble(t, 'error');

  /* ── Typing indicator ────────────────────────── */
  function showTyping() {
    typing.classList.add('is-visible');
    typing.setAttribute('aria-hidden', 'false');
    messages.scrollTop = messages.scrollHeight;
  }
  function hideTyping() {
    typing.classList.remove('is-visible');
    typing.setAttribute('aria-hidden', 'true');
  }

  /* ── Auto-grow textarea ──────────────────────── */
  input.addEventListener('input', () => {
    input.style.height = 'auto';
    input.style.height = Math.min(input.scrollHeight, 110) + 'px';
  });

  /* ── Send on Enter (Shift+Enter = newline) ───── */
  input.addEventListener('keydown', e => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      if (!isWaiting) form.requestSubmit();
    }
  });

  /* ── Submit ──────────────────────────────────── */
  form.addEventListener('submit', async e => {
    e.preventDefault();
    const text = input.value.trim();
    if (!text || isWaiting) return;

    /* Lock UI */
    isWaiting = true;
    sendBtn.disabled = true;
    input.value = '';
    input.style.height = 'auto';

    /* Show user bubble */
    appendUser(text);

    /* Add to history */
    history.push({ role: 'user', content: text });

    /* Show typing */
    showTyping();

    try {
      const res = await fetch(API_URL, {
        method:  'POST',
        headers: { 'Content-Type': 'application/json' },
        body:    JSON.stringify({ message: text, history: history.slice(-10) }),
      });

      if (!res.ok) throw new Error('HTTP ' + res.status);

      const data = await res.json();
      const reply = (data.reply || '').trim();

      if (!reply) throw new Error('empty reply');

      hideTyping();
      appendBot(reply);
      history.push({ role: 'assistant', content: reply });

    } catch (err) {
      hideTyping();
      appendError(ERROR_MSG);
      console.error('[Alessia]', err);
      /* Remove last user message from history so user can retry */
      history.pop();
    } finally {
      isWaiting = false;
      sendBtn.disabled = false;
      input.focus();
    }
  });
})();
</script>
